步骤 2 · 门面

fn extract_isolines(layout: &Layout, cfg: &BranchCalcConfig) -> Vec<IsolineInfo> {
    if cfg.use_pml_split {
        match try_pml_split(layout) {
            Ok(v) => return v,
            Err(issues) => {
                log_issues(issues);
                // fall through to legacy
            }
        }
    }
    extract_isolines_legacy(layout, cfg)
}

步骤 3 · try_pml_split

fn try_pml_split(layout: &Layout) -> Result<Vec<IsolineInfo>, Vec<SplitIssue>> {
    let members = map_to_split_members(layout)?;  // Missing geom → Err
    let r = split_isolines_pml(&members);
    if !r.issues.is_empty() && should_abort(&r.issues) {
        return Err(r.issues);
    }
    Ok(bags_to_isolines(layout, &r.allarr, &r.ldirs))
}
