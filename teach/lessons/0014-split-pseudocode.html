split_isolines_pml 伪代码

对照课 1 决策树 + fixture 阈值，把 P1 弯头主路径写成可直接翻译成 Rust 的伪代码。 本课仍不修改 src/mbd。

范围：Head/Tail/通过件 + 弯头三条岔路 + 桥接。 不做：PCOM / special / Polar 邻段（P3/P4）。

类型（与 P0 对齐）

MemberKind = Head|Tail|Tubi|Atta|Weld|Olet|Elbo|Bend|Pcom|Other
PASS_THROUGH = {Tubi, Atta, Weld, Olet}

IsolineBag = { mem_refnos: [string], leave_dir: Vec3 }

fn angle_deg(a, b) -> degrees in [0, 180]
fn dist(a, b) -> mm

主循环伪代码

fn split_isolines_pml(members: [BranchMemberGeom]) -> Result<[IsolineBag], Issues>:
  allarr = []
  onearr = []
  poss = []          // turn points
  ldirs = []         // leave dir per open/closed segment
  issues = []
  run_ldir = None    // running segment direction

  for m in members sorted by order:
    if m.kind == Head:
      onearr = ["HEAD"]  // or m.refno
      poss.append(m.arrive_pos or m.pos)
      run_ldir = m.leave_dir
      ldirs.append(run_ldir)
      continue

    if m.kind == Tail:
      onearr.append(m.refno)
      allarr.append(onearr)
      onearr = []
      continue

    if m.kind in PASS_THROUGH:
      onearr.append(m.refno)
      continue

    if m.kind == Pcom:
      issues.push("P1: PCOM not implemented")
      onearr.append(m.refno)   // or skip — pick one policy & lock it
      continue

    // --- bend / direction-change path ---
    if m.leave_dir is None:
      issues.push("missing leave_dir: " + m.refno)
      continue

    dir = m.leave_dir
    if run_ldir is None:
      run_ldir = dir
      onearr.append(m.refno)
      continue

    if angle_deg(dir, run_ldir) <= 0.01:
      onearr.append(m.refno)
      continue

    // entered dir-change (> 0.01°)
    adir = m.arrive_dir
    apos = m.arrive_pos
    lpos = m.leave_pos
    memldir = dir
    turn = m.pos or midpoint(apos, lpos)

    // ① near-180 correction
    if adir and apos and poss.last and
       angle_deg(adir, memldir) > 179.99 and
       angle_deg(adir, run_ldir) > 179 and
       dist(apos, poss.last) < 1:
         run_ldir = opposite(adir)
         ldirs[last] = run_ldir
         continue

    // ② main split
    need = (apos and lpos and dist(apos, lpos) > 1)
            or angle_deg(dir, run_ldir) > 10
    if need:
      if lpos and poss.last and dist(lpos, poss.last) < 0.1:
        ldirs[last] = dir
        run_ldir = dir
        onearr.append(m.refno)
      else:
        poss.append(turn)
        onearr.append(m.refno)       // bridge: end of previous
        allarr.append(onearr)
        run_ldir = dir
        onearr = [m.refno]           // bridge: start of next
        ldirs.append(run_ldir)

    // ③ force split if adir∠memldir > 100
    if adir and angle_deg(adir, memldir) > 100:
      // same bridge close/open as ②-else (may double-split; match PML)
      poss.append(turn)
      if onearr not empty and onearr.last != m.refno:
        onearr.append(m.refno)
      allarr.append(onearr)
      run_ldir = dir
      onearr = [m.refno]
      ldirs.append(run_ldir)
    else if not need:
      onearr.append(m.refno)

  if onearr not empty:
    issues.push("unterminated bag without Tail")
  return (allarr as IsolineBags, issues)

对 fixture 的自检

  • toy-bridge-and-stick：② 在 EA 走 else 桥接；EB 走 stick 分支
  • colinear-no-split：全程 ≤0.01，不进变向
  • variant-eb-no-stick：EB 也走桥接 → 3 袋

技能：读伪代码
